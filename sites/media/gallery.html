Accessing images

Javascript example 

fetch('{{smartdataurlnenc}}/smartdata/records/{{collection_media}}/?storage=smartmonitoring&filter=ts%2Cgt%2C2022-12-06T00%3A00&filter=ts%2Clt%2C2022-12-06T23%3A59&order=ts', {
    method: "GET"
})
.then(response => response.json())
.then(data => {

    // Base URL where image files are served (adjust if necessary)
    const baseImageUrl = "{{baseurl}}/";

    data.records.forEach(record => {

        // The relative image path stored in the dataset
        const relativePath = record.path;

        // Build the full HTTP URL to the image
        const imageUrl = baseImageUrl.replace(/\/+$/, "") + "/" + relativePath.replace(/^\/+/, "");

        console.log("Image URL:", imageUrl);

        // Example: create an  element and display the image
        // const img = document.createElement("img");
        // img.src = imageUrl;
        // document.body.appendChild(img);
    });
})
.catch(err => console.error("Error fetching data:", err));

Python example 

import requests

# API endpoint for retrieving records
url = (
    "{{smartdataurlnenc}}"
    "smartdata/records/{{collection_media}}"
    "?storage=smartmonitoring"
    "&filter=ts%2Cgt%2C2022-12-06T00%3A00"
    "&filter=ts%2Clt%2C2022-12-06T23%3A59"
    "&order=ts"
)

response = requests.get(url)
data = response.json()

# Base URL where image files are served (adjust if necessary)
base_image_url = "{{baseurl}}"

def join_url(base, relative):
    """Safely join base URL and relative path without duplicate slashes."""
    return base.rstrip("/") + "/" + relative.lstrip("/")

for record in data.get("records", []):
    # Relative path stored in the dataset
    relative_path = record.get("path")

    # Build the full HTTP URL to the image
    image_url = join_url(base_image_url, relative_path)

    print("Image URL:", image_url)
